<script type="module">

import * as pdfjsLib from

  "https://cdn.jsdelivr.net/npm/pdfjs-dist@4.10.38/build/pdf.min.mjs";


pdfjsLib.GlobalWorkerOptions.workerSrc =

  "https://cdn.jsdelivr.net/npm/pdfjs-dist@4.10.38/build/pdf.worker.min.mjs";


const pdfPath = "./manual.pdf";


const pageA = document.getElementById("pageA");

const pageB = document.getElementById("pageB");

const pages = document.getElementById("pages");

const counter = document.getElementById("counter");

const message = document.getElementById("message");

const prev = document.getElementById("prev");

const next = document.getElementById("next");


let pdf;

let groups = [];

let position = 0;

let renderId = 0;


function makeGroups(total) {

  const result = [[1]];

  let p = 2;


  // 表紙を除き、最後の2ページは単独表示

  while (total - p + 1 > 2) {

    result.push([p, p + 1]);

    p += 2;

  }


  while (p <= total) {

    result.push([p]);

    p++;

  }


  return result;

}


async function renderPage(number, container, id) {

  container.innerHTML = "";

  container.classList.remove("empty");


  const page = await pdf.getPage(number);

  if (id !== renderId) return;


  const base = page.getViewport({ scale: 1 });

  const mobile = window.innerWidth <= 650;


  const availableWidth = container.clientWidth - 12;

  const availableHeight = container.clientHeight - 12;


  const scale = Math.min(

    availableWidth / base.width,

    availableHeight / base.height

  );


  const viewport = page.getViewport({ scale });


  const canvas = document.createElement("canvas");

  const context = canvas.getContext("2d");


  const ratio = window.devicePixelRatio || 1;


  canvas.width = Math.floor(viewport.width * ratio);

  canvas.height = Math.floor(viewport.height * ratio);

  canvas.style.width = viewport.width + "px";

  canvas.style.height = viewport.height + "px";


  container.appendChild(canvas);


  await page.render({

    canvasContext: context,

    viewport: viewport,

    transform: [ratio, 0, 0, ratio, 0, 0]

  }).promise;

}


async function renderSpread() {

  if (!pdf) return;


  const id = ++renderId;

  const group = groups[position];

  const mobile = window.innerWidth <= 650;


  pages.classList.toggle(

    "spread",

    group.length === 2

  );


  pageA.innerHTML = "";

  pageB.innerHTML = "";

  pageB.classList.remove("empty");


  await renderPage(group[0], pageA, id);

  if (id !== renderId) return;


  if (group.length === 2 && !mobile) {

    await renderPage(group[1], pageB, id);

  } else {

    pageB.classList.add("empty");

  }


  if (id !== renderId) return;


  const first = group[0];

  const last = group[group.length - 1];


  counter.textContent =

    `${first}${first !== last ? "–" + last : ""} / ${pdf.numPages}`;


  prev.disabled = position === 0;

  next.disabled = position === groups.length - 1;

}


function changePage(direction) {

  const nextPosition = position + direction;


  if (nextPosition < 0 || nextPosition >= groups.length) {

    return;

  }


  position = nextPosition;

  renderSpread();

}


prev.addEventListener("click", () => changePage(-1));

next.addEventListener("click", () => changePage(1));


document.addEventListener("keydown", event => {

  if (event.key === "ArrowLeft") changePage(-1);

  if (event.key === "ArrowRight") changePage(1);

});


let resizeTimer;

window.addEventListener("resize", () => {

  clearTimeout(resizeTimer);

  resizeTimer = setTimeout(renderSpread, 150);

});


async function init() {

  try {

    pdf = await pdfjsLib.getDocument(pdfPath).promise;

    groups = makeGroups(pdf.numPages);


    message.style.display = "none";

    position = 0;


    await renderSpread();

  } catch (error) {

    message.textContent =

      "Manual not found. Please upload manual.pdf.";

    console.error(error);

  }

}


init();
